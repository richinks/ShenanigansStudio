<script>
  import { onMount } from 'svelte'
  import { supabase } from '#lib/supabase'

  let setlists = [], loading = true, err = ''
  let showCreate = false, name = '', gig_date = '', venue = '', notes = '', saving = false

  onMount(async () => {
    await load()
  })

  async function load() {
    loading = true
    const { data, error } = await supabase
      .from('setlists')
      .select('id, name, gig_date, venue, notes, created_at, setlist_songs(count)')
      .order('gig_date', { ascending: false })
    if (error) err = error.message
    setlists = data ?? []
    loading = false
  }

  async function create() {
    if (!name.trim()) return
    saving = true
    const { error } = await supabase.from('setlists').insert({ name, gig_date: gig_date || null, venue: venue || null, notes: notes || null })
    saving = false
    if (error) { err = error.message; return }
    name = ''; gig_date = ''; venue = ''; notes = ''
    showCreate = false
    await load()
  }

  function fmt(d) {
    if (!d) return '—'
    return new Date(d + 'T00:00:00').toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' })
  }
</script>

<div class="header">
  <h1>Setlists</h1>
  <button on:click={() => showCreate = !showCreate}>＋ New Setlist</button>
</div>

{#if showCreate}
  <div class="form">
    <input bind:value={name} placeholder="Setlist name *" />
    <input bind:value={gig_date} type="date" />
    <input bind:value={venue} placeholder="Venue" />
    <input bind:value={notes} placeholder="Notes" />
    <div class="form-actions">
      <button on:click={create} disabled={saving || !name.trim()}>{saving ? 'Saving…' : 'Create'}</button>
      <button class="ghost" on:click={() => showCreate = false}>Cancel</button>
    </div>
  </div>
{/if}

{#if err}<p style="color:#f87171">⚠️ {err}</p>{/if}

{#if loading}
  <p class="muted">Loading…</p>
{:else if setlists.length === 0}
  <p class="muted">No setlists yet. Create one above.</p>
{:else}
  <div class="grid">
    {#each setlists as sl}
      <a href="/setlists/{sl.id}" class="card">
        <div class="card-top">
          <span class="card-name">{sl.name}</span>
          <span class="badge">{sl.setlist_songs?.[0]?.count ?? 0} songs</span>
        </div>
        <div class="card-meta">
          {#if sl.gig_date}<span>📅 {fmt(sl.gig_date)}</span>{/if}
          {#if sl.venue}<span>📍 {sl.venue}</span>{/if}
        </div>
        {#if sl.notes}<p class="card-notes">{sl.notes}</p>{/if}
      </a>
    {/each}
  </div>
{/if}
