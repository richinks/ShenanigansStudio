# Auto-switch Node version when entering ShenanigansStudio PWA
function Set-NodeVersion {
    param([string]$path)

    if ($path -like "D:\ShenanigansStudio\pwa*") {
        Write-Host "Switching to Node 22.17.0 for ShenanigansStudio PWA..." -ForegroundColor Cyan
        nvm use 22.17.0 | Out-Null
    }
}

# Hook into directory changes
Register-EngineEvent PowerShell.OnLocationChanged -Action {
    Set-NodeVersion (Get-Location).Path
} | Out-Null
